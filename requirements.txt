import pandas as pd
import streamlit as st
from decimal import Decimal

APP_TITLE = "Mizan | ميزان"
FOUNDER_NAME = "جنى محمود"

st.set_page_config(page_title=f"{APP_TITLE} - {FOUNDER_NAME}", page_icon="⚖️", layout="wide")

st.markdown("""
    <style>
    @import url('https://fonts.googleapis.com/css2?family=Cairo:wght@700&display=swap');
    html, body, [class*="css"] { font-family: 'Cairo', sans-serif !important; }
    .main-header { background: linear-gradient(135deg, #0f172a 0%, #1e293b 100%); color: white; padding: 2rem; border-radius: 16px; margin-bottom: 2rem; text-align: center; }
    </style>
""", unsafe_allow_html=True)

def safe_to_decimal(val):
    try:
        if pd.isna(val) or val == "": return None
        return Decimal(str(val).replace(',',''))
    except: return None

# Sidebar
st.sidebar.title(f"{APP_TITLE} ⚖️")
st.sidebar.caption(f"تطوير: {FOUNDER_NAME}")
menu = st.sidebar.radio("القائمة:", ["الصفحة الرئيسية", "رفع وفحص الملفات"])

if menu == "الصفحة الرئيسية":
    st.markdown(f"""<div class="main-header"><h1>{APP_TITLE}</h1><p>نظام الذكاء المحاسبي لاكتشاف الأخطاء - فكرة وتطوير: {FOUNDER_NAME}</p><p>ميزانك يظبط في ثانية ⚖️</p></div>""", unsafe_allow_html=True)
    st.success("مشروعك جاهز! اذهبي لتبويب رفع وفحص الملفات لتجربة النظام")
    if st.button("🚀 تجربة بيانات تجريبية"):
        demo = {"البيان":["مبيعات","مشتريات","كهرباء"], "مدين":[0,5000,400], "دائن":[10000,0,0], "الضريبة":[1200,700,0]}
        st.session_state.df = pd.DataFrame(demo)
        st.write(st.session_state.df)

elif menu == "رفع وفحص الملفات":
    st.title("🔍 رفع وفحص ملفات Excel")
    file = st.file_uploader("اختر ملف Excel", type=["xlsx","xls"])
    if file:
        df = pd.read_excel(file)
        st.session_state.df = df
        st.dataframe(df.head())
    
    if 'df' in st.session_state:
        df = st.session_state.df
        cols = ["-- غير محدد --"] + list(df.columns)
        debit_c = st.selectbox("عمود المدين:", cols)
        credit_c = st.selectbox("عمود الدائن:", cols)
        
        if st.button("⚡ بدء الفحص", type="primary"):
            issues = []
            if debit_c != "-- غير محدد --" and credit_c != "-- غير محدد --":
                total_d = sum([safe_to_decimal(x) or 0 for x in df[debit_c]])
                total_c = sum([safe_to_decimal(x) or 0 for x in df[credit_c]])
                st.metric("إجمالي المدين", float(total_d))
                st.metric("إجمالي الدائن", float(total_c))
                st.metric("الفرق", float(abs(total_d - total_c)))
                if total_d != total_c:
                    st.error(f"الميزان غير متوازن! الفرق {abs(total_d-total_c)}")
                    issues.append("عدم توازن المدين والدائن")
                else:
                    st.success("الميزان متوازن ✅")
            
            for idx, row in df.iterrows():
                d = safe_to_decimal(row.get(debit_c)) if debit_c != "-- غير محدد --" else None
                c = safe_to_decimal(row.get(credit_c)) if credit_c != "-- غير محدد --" else None
                if d is None and c is None:
                    st.warning(f"صف {idx+2}: قيمة مفقودة في المدين والدائن")
                    issues.append(f"صف {idx+2} مفقود")
            
            if not issues:
                st.balloons()
                st.success("لا توجد أخطاء! الملف سليم 100%")
